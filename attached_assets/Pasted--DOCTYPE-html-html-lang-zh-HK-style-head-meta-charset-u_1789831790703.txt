  (function() {
    let glucose = 10.2;
    let timing = 'postmeal';
    const displayEl = document.getElementById('glucoseValueDisplay');
    const badgeEl = document.getElementById('statusBadge');
    const pointerEl = document.getElementById('rangeIndicatorPointer');
    const adviceBoxEl = document.getElementById('healthAdviceBox');
    const btnMinus = document.getElementById('btnMinus');
    const btnPlus = document.getElementById('btnPlus');
    const decimalBtns = document.querySelectorAll('.decimal-btn');
    const timingChips = document.querySelectorAll('.meal-timing-chip');
    const tagChips = document.querySelectorAll('.tag-chip');
    const saveBtn = document.getElementById('saveGlucoseBtn');

    function updateUI() {
      displayEl.textContent = glucose.toFixed(1);
      const dec = Math.round((glucose - Math.floor(glucose)) * 10);
      decimalBtns.forEach(btn => {
        const d = parseInt(btn.getAttribute('data-decimal'), 10);
        if (d === dec) {
          btn.className = "decimal-btn active-decimal min-h-[48px] rounded-xl bg-primary text-on-primary font-label-lg text-label-lg font-bold shadow transition-all active:scale-95";
        } else {
          btn.className = "decimal-btn min-h-[48px] rounded-xl bg-surface-container hover:bg-surface-container-high text-on-surface font-label-lg text-label-lg font-bold transition-all active:scale-95";
        }
      });

      let percent = Math.min(Math.max(((glucose - 3.0) / 11.0) * 100, 4), 96);
      pointerEl.style.left = percent + '%';

      if (timing === 'fasting') {
        if (glucose < 3.9) {
          badgeEl.textContent = '偏低';
          badgeEl.className = 'font-bold text-tertiary';
          adviceBoxEl.innerHTML = '<span class="material-symbols-outlined text-tertiary text-[24px] mt-0.5">warning</span><p class="font-body-md text-body-md text-on-surface leading-snug">空腹血糖偏低，可適當含一粒糖果或半杯果汁，稍後復測。</p>';
        } else if (glucose <= 6.1) {
          badgeEl.textContent = '理想正常';
          badgeEl.className = 'font-bold text-primary';
          adviceBoxEl.innerHTML = '<span class="material-symbols-outlined text-primary text-[24px] mt-0.5" style="font-variation-settings: \'FILL\' 1;">check_circle</span><p class="font-body-md text-body-md text-on-surface leading-snug">空腹數值非常理想！繼續保持良好的飲食作息習慣。</p>';
        } else {
          badgeEl.textContent = '偏高';
          badgeEl.className = 'font-bold text-secondary';
          adviceBoxEl.innerHTML = '<span class="material-symbols-outlined text-secondary text-[24px] mt-0.5">info</span><p class="font-body-md text-body-md text-on-surface leading-snug">早晨空腹偏高，多飲溫水，留意昨晚晚餐澱粉質攝取。</p>';
        }
      } else {
        if (glucose < 4.4) {
          badgeEl.textContent = '偏低';
          badgeEl.className = 'font-bold text-tertiary';
          adviceBoxEl.innerHTML = '<span class="material-symbols-outlined text-tertiary text-[24px] mt-0.5">warning</span><p class="font-body-md text-body-md text-on-surface leading-snug">血糖偏低，請先進食少量餅乾或高纖點心，如有頭暈請坐低休息。</p>';
        } else if (glucose <= 8.5) {
          badgeEl.textContent = '理想標準';
          badgeEl.className = 'font-bold text-primary';
          adviceBoxEl.innerHTML = '<span class="material-symbols-outlined text-primary text-[24px] mt-0.5" style="font-variation-settings: \'FILL\' 1;">check_circle</span><p class="font-body-md text-body-md text-on-surface leading-snug">飯後血糖控制得好好，維持均衡餐盤同散步習慣！</p>';
        } else {
          badgeEl.textContent = '稍微偏高';
          badgeEl.className = 'font-bold text-secondary';
          adviceBoxEl.innerHTML = '<span class="material-symbols-outlined text-secondary text-[24px] mt-0.5">info</span><p class="font-body-md text-body-md text-on-surface leading-snug">飯後數值稍微偏高，多飲溫水放鬆休息下，半小時後再量一次。</p>';
        }
      }
    }

    btnMinus.addEventListener('click', function() {
      if (glucose > 2.0) {
        glucose = Math.round((glucose - 0.1) * 10) / 10;
        updateUI();
      }
    });

    btnPlus.addEventListener('click', function() {
      if (glucose < 25.0) {
        glucose = Math.round((glucose + 0.1) * 10) / 10;
        updateUI();
      }
    });

    decimalBtns.forEach(btn => {
      btn.addEventListener('click', function() {
        const decVal = parseInt(this.getAttribute('data-decimal'), 10);
        const intVal = Math.floor(glucose);
        glucose = intVal + (decVal / 10);
        updateUI();
      });
    });

    timingChips.forEach(btn => {
      btn.addEventListener('click', function() {
        timingChips.forEach(c => {
          c.className = "meal-timing-chip flex flex-col items-center justify-center min-h-[58px] py-2 px-1 rounded-xl bg-surface-container text-on-surface-variant font-label-md text-label-md transition-all active:scale-98";
        });
        this.className = "meal-timing-chip active-timing flex flex-col items-center justify-center min-h-[58px] py-2 px-1 rounded-xl bg-primary text-on-primary font-label-md text-label-md shadow-md transition-all active:scale-98";
        timing = this.getAttribute('data-timing');
        updateUI();
      });
    });

    tagChips.forEach(chip => {
      chip.addEventListener('click', function() {
        const isActive = this.classList.contains('bg-primary');
        if (isActive) {
          this.className = "tag-chip flex items-center gap-1.5 min-h-[48px] px-3.5 rounded-xl bg-surface-container text-on-surface-variant font-label-md text-label-md transition-all active:scale-95";
        } else {

        }
      });

    });

    saveBtn.addEventListener('click', function() {
      this.classList.add('scale-95');
      setTimeout(() => {
        this.classList.remove('scale-95');
        alert('血糖記錄已成功儲存：' + glucose.toFixed(1) + ' mmol/L');
      }, 150);
    });

    updateUI();
  })();

<script data-purpose="audio-simulation">
    function triggerVoicePrompt() {
      const btn = document.getElementById('listenBtn');
      if (btn.classList.contains('bg-brand-teal')) {
        btn.classList.remove('bg-brand-teal', 'text-white');
        btn.classList.add('bg-white', 'text-brand-teal');
        btn.innerHTML = '<span>▶ 聽建議</span>';
      } else {
        btn.classList.add('bg-brand-teal', 'text-white');
        btn.classList.remove('bg-white', 'text-brand-teal');
        btn.innerHTML = '<span>⏸ 播放中...</span>';
        
        // Simple Web Speech API fallback in Cantonese / Chinese if available
        if ('speechSynthesis' in window) {
          const utterance = new SpeechSynthesisUtterance("下次可以試下：加點優質蛋白質，例如鮮魚、滑豆腐、水煮蛋或者瘦肉。");
          utterance.lang = 'zh-HK';
          utterance.rate = 0.9; // Slightly slower for elderly comprehension
          utterance.onend = () => {
            btn.classList.remove('bg-brand-teal', 'text-white');
            btn.classList.add('bg-white', 'text-brand-teal');
            btn.innerHTML = '<span>▶ 聽建議</span>';
          };
          window.speechSynthesis.speak(utterance);
        }
      }
    }
  </script>
