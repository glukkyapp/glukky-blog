function toInternalId(s: string): string {
  return s.trim().toLowerCase()
    .replace(/\([^)]*\)/g, "")    // delete "(sweet soy)" entirely
    .replace(/[^a-z0-9]+/g, "_")  // spaces/symbols → underscore
    .replace(/^_+|_+$/g, "");     // trim leading/trailing underscores
}

function parseSauceIds(saucesEn: string | null): string[] {
  if (!saucesEn) return [];
  return saucesEn.split(",").map(s => toInternalId(s)).filter(Boolean);
}

function parseToppingIds(toppingsEn: string | null): string[] {
  if (!toppingsEn) return [];
  return toppingsEn.split(",").map(s => toInternalId(s)).filter(Boolean);
}
