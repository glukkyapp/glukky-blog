tsx
type GardenAsset =
  | "background" | "seedling" | "small-tree" | "young-tree" | "complete-tree"
  | "flower1" | "flower2" | "flower3" | "flower4" | "bird" | "ferry" | "bench" | "lamp";

const treeForPoints = (points: number) => {
  if (points >= 31) return "complete-tree";
  if (points >= 16) return "young-tree";
  if (points >= 7) return "small-tree";
  if (points >= 1) return "seedling";
  return null;
};

const stackedAssets = [
  { asset: "flower1", minimumPoints: 7 },
  { asset: "flower2", minimumPoints: 16 },
  { asset: "bird", minimumPoints: 16 },
  { asset: "bench", minimumPoints: 31 },
  { asset: "lamp", minimumPoints: 31 },
  { asset: "flower3", minimumPoints: 46 },
  { asset: "ferry", minimumPoints: 46 },
  { asset: "flower4", minimumPoints: 60 },
];

export function HarbourGardenSvg({ points }: { points: number }) {
  const tree = treeForPoints(points);
  return (
    <div className="relative aspect-[16/9] w-full overflow-hidden"
         aria-label={`Harbour Garden with ${points} points`}>
      <img src="/garden/background.png" alt="" className="absolute inset-0 h-full w-full object-contain" />
      {tree && (
        <img src={`/garden/${tree}.png`} alt="" className="absolute inset-0 h-full w-full object-contain" />
      )}
      {stackedAssets
        .filter((item) => points >= item.minimumPoints)
        .map((item) => (
          <img key={item.asset} src={`/garden/${item.asset}.png`} alt=""
               className="absolute inset-0 h-full w-full object-contain" />
        ))}
    </div>
  );
}
