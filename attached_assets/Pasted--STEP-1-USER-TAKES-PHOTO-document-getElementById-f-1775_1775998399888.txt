// ─── STEP 1: USER TAKES PHOTO ─────────────────────────────────────────
document.getElementById('foodCamera').addEventListener('change', async (e) => {
  const imageFile = e.target.files[0];
  const result = await processFoodPhoto(imageFile, currentUserId);
  fillFormAndAwaitConfirmation(result);
});


// ─── STEP 2: MAIN PIPELINE ───────────────────────────────────────────
async function processFoodPhoto(imageFile, userId) {
  const imageElement = await fileToImageElement(imageFile);

  // 2a. Run TM model locally (free, no API call)
  const predictions = await tmModel.predict(imageElement);
  const top = predictions.sort((a, b) => b.probability - a.probability)[0];

  let foodName, source;

  if (top.probability >= 0.85) {
    // TM is confident → use its result
    foodName = top.className;
    source = "TM_local";
  } else {
    // TM not confident → call Claude Sonnet Vision as fallback
    const claudeResult = await callClaudeVision(imageFile);
    foodName = claudeResult.food_name;
    source = "Claude_fallback";
  }

  // 2b. Get default combo from server database
  const defaultCombo = await getComboFromDB(foodName); // fetch from combos.json

  // 2c. Apply user preferences on top of defaults (only fields with modifyCount >= 2)
  const personalised = await applyUserPreferences(userId, foodName, defaultCombo);

  return { foodName, ...personalised, source };
}


// ─── STEP 3: APPLY USER PREFERENCES ──────────────────────────────────
async function applyUserPreferences(userId, foodName, defaultCombo) {
  const prefs = await db.getUserPref(userId, foodName);
  // Start with the default combo
  const result = { ...defaultCombo };

  if (!prefs) return result; // no saved prefs → return default as-is

  // Only override fields the user has consistently modified (2+ times)
  const THRESHOLD = 2;
  if (prefs.portion.modifyCount  >= THRESHOLD) result.portion  = prefs.portion.value;
  if (prefs.sauce.modifyCount    >= THRESHOLD) result.sauce    = prefs.sauce.value;
  if (prefs.toppings.modifyCount >= THRESHOLD) result.toppings = prefs.toppings.value;

  return result;
}


// ─── STEP 4: USER CONFIRMS OR EDITS ──────────────────────────────────
// UI shows pre-filled form with chips for portion / sauce / toppings
// User taps Confirm, or adjusts 1-3 chips then taps Confirm

async function onUserConfirmed(userId, foodName, originalDefaults, finalEntry) {
  // Save the food log entry to diary
  await db.saveFoodLog(userId, finalEntry);

  // Track any fields the user changed from what was shown
  await trackUserModifications(userId, foodName, originalDefaults, finalEntry);
}


// ─── STEP 5: TRACK MODIFICATIONS → UPDATE PREFERENCES ────────────────
async function trackUserModifications(userId, foodName, shown, confirmed) {
  const prefs = await db.getUserPref(userId, foodName) || {
    portion:  { value: shown.portion,  modifyCount: 0 },
    sauce:    { value: shown.sauce,    modifyCount: 0 },
    toppings: { value: shown.toppings, modifyCount: 0 }
  };

  // Check each field: did the user change it?
  if (confirmed.portion !== shown.portion) {
    prefs.portion.value = confirmed.portion;
    prefs.portion.modifyCount += 1;
  }

  if (JSON.stringify(confirmed.sauce) !== JSON.stringify(shown.sauce)) {
    prefs.sauce.value = confirmed.sauce;
    prefs.sauce.modifyCount += 1;
  }

  if (JSON.stringify(confirmed.toppings) !== JSON.stringify(shown.toppings)) {
    prefs.toppings.value = confirmed.toppings;
    prefs.toppings.modifyCount += 1;
  }

  // Save updated preferences back to DB
  await db.saveUserPref(userId, foodName, prefs);
}


// ─── STEP 6: CLAUDE FALLBACK (only when TM < 85% confident) ──────────
async function callClaudeVision(imageFile) {
  const base64Image = await compressToBase64(imageFile, 400);

  const response = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: {
      "x-api-key": CLAUDE_API_KEY,
      "anthropic-version": "2023-06-01",
      "anthropic-beta": "prompt-caching-2024-07-31",
      "content-type": "application/json"
    },
    body: JSON.stringify({
      model: "claude-sonnet-4-5",
      max_tokens: 100,
      system: [{
        type: "text",
        text: "You are a Hong Kong food expert. Identify the dish name only. Return JSON: { food_name: string (Traditional Chinese) }",
        cache_control: { type: "ephemeral" }
      }],
      messages: [{
        role: "user",
        content: [
          { type: "image", source: { type: "base64", media_type: "image/jpeg",  base64Image } },
          { type: "text", text: "What HK dish is this? Return JSON only." }
        ]
      }]
    })
  });

  const data = await response.json();
  return JSON.parse(data.content[0].text);
  // Returns food_name only → server then looks up combos.json for the rest
}
